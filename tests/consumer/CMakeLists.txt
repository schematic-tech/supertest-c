cmake_minimum_required(VERSION 3.20)
project(SupertestConsumer LANGUAGES C)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
if(DEFINED SUPERTEST_SOURCE)
    add_subdirectory("${SUPERTEST_SOURCE}" supertest)
elseif(DEFINED SUPERTEST_ARCHIVE)
    include(FetchContent)
    FetchContent_Declare(supertest URL "${SUPERTEST_ARCHIVE}" URL_HASH "SHA256=${SUPERTEST_ARCHIVE_SHA256}")
    FetchContent_MakeAvailable(supertest)
else()
    find_package(SchematicSupertest 0.1.0 CONFIG REQUIRED)
endif()
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE schematic::supertest)
enable_testing()
add_test(NAME consumer COMMAND consumer)
